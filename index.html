function showSettleReportModal(collectorName, expectedAmount, actualAmount, diffText, records) { let content = `شركة العالمية - سند تسوية: ${collectorName}\nالتاريخ: ${new Date().toLocaleDateString()}\n-------------------\n`; records.forEach(r => { let disc = r.discount ? ` | حسم: $${r.discount}` : ''; content += `${r.code} | ${r.type} | $${r.amount}${disc}\n`; }); content += `-------------------\nالمتوقع: $${expectedAmount.toFixed(2)}\nالفعلي: $${actualAmount.toFixed(2)}\nالنتيجة: ${diffText}`; document.getElementById("settleDetailsContent").innerHTML = content.replace(/\n/g, '
'); // إضافة زر مشاركة حقيقي document.getElementById("settleDetailsContent").innerHTML += `
`; document.getElementById("settleModal").style.display = "flex"; // تفعيل زر المشاركة document.getElementById("shareBtn").onclick = () => { if (navigator.share) { navigator.share({ title: 'سند تسوية - شركة العالمية', text: content }).catch(err => alert("خطأ في المشاركة")); } else { alert("جهازك لا يدعم المشاركة المباشرة، يرجى أخذ لقطة شاشة."); } }; }
